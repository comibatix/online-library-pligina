import React, { createContext, useState, useContext, useEffect, ReactNode } from 'react';
import axios from 'axios';

interface User {
  id: string;
  login: string;
}

interface AuthContextType {
  user: User | null;
  token: string | null;
  login: (login: string, password: string) => Promise<void>;
  register: (login: string, password: string) => Promise<void>;
  logout: () => void;
  isAuthenticated: boolean;
}

const AuthContext = createContext<AuthContextType | undefined>(undefined);

export const AuthProvider: React.FC<{ children: ReactNode }> = ({ children }) => {
  const [user, setUser] = useState<User | null>(null);
  const [token, setToken] = useState<string | null>(null);
  const [isLoaded, setIsLoaded] = useState(false);

  // Проверяем токен только при первом монтировании
  useEffect(() => {
    const storedToken = localStorage.getItem('token');
    if (storedToken) {
      axios.get('/api/profile', {
        headers: { Authorization: `Bearer ${storedToken}` }
      })
      .then(res => {
        setToken(storedToken);
        setUser({ id: res.data.userId, login: res.data.login });
      })
      .catch(() => {
        localStorage.removeItem('token');
      })
      .finally(() => {
        setIsLoaded(true);
      });
    } else {
      setIsLoaded(true);
    }
  }, []);

  const login = async (loginStr: string, password: string) => {
    const res = await axios.post('/api/auth/login', { login: loginStr, password });
    const newToken = res.data.token;
    const newUser = res.data.user;
    localStorage.setItem('token', newToken);
    setToken(newToken);
    setUser(newUser);
  };

  const register = async (loginStr: string, password: string) => {
    await axios.post('/api/auth/register', { login: loginStr, password });
    await login(loginStr, password);
  };

  const logout = () => {
    localStorage.removeItem('token');
    setToken(null);
    setUser(null);
  };

  // Пока не загрузили данные из localStorage, не рендерим детей
  if (!isLoaded) {
    return null;
  }

  return (
    <AuthContext.Provider value={{ user, token, login, register, logout, isAuthenticated: !!token }}>
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error('useAuth должен использоваться внутри AuthProvider');
  }
  return context;
};
